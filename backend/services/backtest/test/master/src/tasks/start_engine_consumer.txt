use crate::{
    state::{AppState, MasterState},
};
use tokio::sync::RwLockWriteGuard;
use tracing::{error, info};

pub fn start_engine_consumer(
    state: AppState,
    mut consumer: /* Pulsar Consumer */,
) {
    tokio::spawn(async move {
        loop {
            //
            // Esperar EngineState desde Pulsar.
            //
            let message = match consumer.receive().await {
                Ok(message) => message,
                Err(error) => {
                    error!(?error, "Failed to receive EngineState.");
                    continue;
                }
            };

            //
            // TODO:
            // Deserializar el payload.
            //
            let engine_state: EngineState = /* deserialize(message.payload()) */;

            //
            // Validar y almacenar.
            //
            {
                let mut master: RwLockWriteGuard<'_, MasterState> =
                    state.master.write().await;

                if master.replay_status != ReplayStatus::Running {
                    continue;
                }

                if engine_state.version != master.version {
                    error!(
                        expected = master.version,
                        received = engine_state.version,
                        "Unexpected EngineState version."
                    );

                    continue;
                }

                if engine_state.tick_index != master.tick_index {
                    error!(
                        expected = master.tick_index,
                        received = engine_state.tick_index,
                        "Unexpected EngineState tick_index."
                    );

                    continue;
                }

                master.engine_state = Some(engine_state);

                info!(
                    tick_index = master.tick_index,
                    "EngineState received."
                );
            }

            //
            // Despertar ReplayTask.
            //
            state.engine_notify.notify_one();

            //
            // Esperar autorización para ACK.
            //
            state.engine_ack_notify.notified().await;

            //
            // Confirmar el mensaje.
            //
            if let Err(error) = consumer.ack(&message).await {
                error!(?error, "Failed to ACK EngineState.");
                continue;
            }

            info!("EngineState ACK.");
        }
    });
}
